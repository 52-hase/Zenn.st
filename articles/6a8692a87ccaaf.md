### app/channels/live_room_channel.rb

class LiveRoomChannel < ApplicationCable::Channel
  def subscribed
    live_room = LiveRoom.find(params[:live_room_id])
    stream_for live_room
  end

  def speak(data)
    live_room = LiveRoom.find_by(id: data["live_room_id"])
    image_data = data["image"]
    message_content = data["message"].presence || ""

    message = live_room.messages.new(content: message_content, user_id: data["user_id"])

    if image_data.present?
      decoded_image = Base64.decode64(image_data)
      message.image.attach(io: StringIO.new(decoded_image), filename: "uploaded_image.png")
    end

    message.save!

    # 左右の表示判定はクライアント側(JS)で行うため、current_user依存の偽装ログインは不要
    template = ApplicationController.renderer.render(
      partial: "messages/message",
      locals: { message: message }
    )
    LiveRoomChannel.broadcast_to(live_room, template)
  end
end

import consumer from "./consumer"

// URLからlive_room_idを取得する関数
function getLiveRoomIdFromUrl() {
  const url = window.location.pathname;
  const match = url.match(/live_rooms\/(\d+)/);
  return match ? match[1] : null;
}

function getUserId() {
  const userIdElement = document.getElementById('user-id');
  return userIdElement ? userIdElement.dataset.userId : null;
}

const liveRoomId = getLiveRoomIdFromUrl();  // URLからlive_room_idを取得
const userId = getUserId();  // ユーザーIDを取得
const App = {};  // Appオブジェクトを定義

if (liveRoomId) {
  App.room = consumer.subscriptions.create({ channel: "LiveRoomChannel", live_room_id: liveRoomId }, {
    connected() {
      console.log('Connected to LiveRoomChannel');
    },

    disconnected() {
      console.log('Disconnected from LiveRoomChannel');
    },

    received: function(message) {
      const messages = document.getElementById('messages');
      if (messages) {
        const ul = messages.querySelector('ul');
        if (ul) {
          // 受け取ったHTMLをリストの最後に追加する
          ul.insertAdjacentHTML('beforeend', message);

          // 追加した要素を取得し、閲覧者本人の視点で左右の位置を判定する
          const lastLi = ul.lastElementChild;
          if (lastLi) {
            const messageUserId = lastLi.dataset.userId;
            const row = lastLi.querySelector('.message-row');
            if (row) {
              if (String(messageUserId) === String(userId)) {
                row.classList.add('justify-content-end');
              } else {
                row.classList.add('justify-content-start');
              }
            }
          }
        }
      }
    },

    speak: function(content, image = null) {
      return this.perform('speak', { message: content, image: image, live_room_id: liveRoomId, user_id: userId });
    }
  });

  document.addEventListener('DOMContentLoaded', function() {
    const input = document.getElementById('chat-input');
    const imageInput = document.getElementById('chat-image');
    const button = document.getElementById('button');
    const chatImageLabel = document.getElementById('chat-image-label');

    // 画像が選択されたときにラベルを更新する
    imageInput.addEventListener('change', function() {
      if (imageInput.files.length > 0) {
        chatImageLabel.textContent = '画像が添付されています';
      }
    });

    button.addEventListener('click', function() {
      const content = input.value;
      const file = imageInput.files[0];

      if (file) {
        const reader = new FileReader();
        reader.onload = function(event) {
          const base64String = event.target.result.split(',')[1];
          App.room.speak(content, base64String);
        };
        reader.readAsDataURL(file);
      } else {
        App.room.speak(content);
      }

      input.value = '';
      imageInput.value = '';
      chatImageLabel.textContent = '画像';  // 入力をクリアした後、ラベルを画像に戻す
    });
  });
}

```javascript
// Action Cableの接続情報を読み込む
import consumer from "./consumer";

// URLからlive_room_idを取得
function getLiveRoomIdFromUrl() {
  // 現在のページのURLのパス部分を取得
  // 例：https://example.com/live_rooms/1
  // → "/live_rooms/1"
  const url = window.location.pathname;

  // URLからlive_room_idの数字部分を取得
  // "/live_rooms/1" → "1"
  const match = url.match(/live_rooms\/(\d+)/);

  // IDが見つかった場合はそのIDを返す
  // 見つからなかった場合はnullを返す
  return match ? match[1] : null;
}

// ログインユーザーIDを取得
function getUserId() {
  // HTMLにある「user-id」というIDの要素を取得
  const userIdElement = document.getElementById("user-id");

  // data-user-idに保存されているユーザーIDを取得
  // 要素が存在しない場合はnullを返す
  return userIdElement ? userIdElement.dataset.userId : null;
}

// メッセージの左右表示を設定
function applyMessagePosition(li) {
  // メッセージの横方向の配置を設定する要素を取得
  const row = li.querySelector(".message-row");

  // message-rowが存在しない場合は処理を終了
  if (!row) return;

  // メッセージのユーザーIDとログインユーザーIDを比較
  if (String(li.dataset.userId) === String(userId)) {
    // 自分のメッセージなら右側に表示
    row.classList.add("justify-content-end");
  } else {
    // 他のユーザーのメッセージなら左側に表示
    row.classList.add("justify-content-start");
  }
}

// 現在のURLからLiveRoomのIDを取得
const liveRoomId = getLiveRoomIdFromUrl();

// HTMLからログインユーザーのIDを取得
const userId = getUserId();

// Action Cableで使用するオブジェクトを作成
const App = {};

// LiveRoomのIDが取得できた場合のみAction Cableに接続
if (liveRoomId) {

  // LiveRoomChannelを購読する
  // これによってLiveRoom内のリアルタイム通信を開始する
  App.room = consumer.subscriptions.create(
    {
      channel: "LiveRoomChannel",
      live_room_id: liveRoomId
    },
    {

      // Action Cableへの接続が成功したときに実行
      connected() {
        console.log("Connected to LiveRoomChannel");
      },

      // Action Cableとの接続が切断されたときに実行
      disconnected() {
        console.log("Disconnected from LiveRoomChannel");
      },

      // Rails側からメッセージを受信したときに実行
      received(message) {
        // メッセージ一覧を表示する要素を取得
        const messages = document.getElementById("messages");

        // messagesが存在する場合のみ処理
        if (messages) {

          // メッセージ一覧のul要素を取得
          const ul = messages.querySelector("ul");

          // ulが存在する場合のみ処理
          if (ul) {

            // Railsから受け取ったHTMLを
            // メッセージ一覧の最後に追加
            ul.insertAdjacentHTML("beforeend", message);

            // 追加された最後のメッセージを取得
            const lastLi = ul.lastElementChild;

            // メッセージが取得できた場合
            if (lastLi) {

              // 自分のメッセージなら右側、
              // 他のユーザーなら左側に配置
              applyMessagePosition(lastLi);
            }
          }
        }
      },

      // メッセージをRails側へ送信する処理
      speak(content, image = null) {

        // LiveRoomChannelのspeakアクションを実行
        return this.perform("speak", {

          // 入力されたメッセージ本文
          message: content,

          // 画像があればBase64形式の画像データ
          // なければnull
          image: image,

          // 現在表示しているLiveRoomのID
          live_room_id: liveRoomId,

          // メッセージを送信したユーザーのID
          user_id: userId
        });
      }
    }
  );

  // HTMLの読み込みが完了したあとに実行
  document.addEventListener("DOMContentLoaded", function () {

    // すでに画面に表示されているメッセージを取得
    // data-user-idを持つli要素が対象
    document
      .querySelectorAll("#messages li[data-user-id]")

      // 取得したすべてのメッセージに対して
      // 自分か他ユーザーかを判定
      .forEach(applyMessagePosition);

    // チャット入力欄を取得
    const input = document.getElementById("chat-input");

    // 画像選択用のinputを取得
    const imageInput = document.getElementById("chat-image");

    // 送信ボタンを取得
    const button = document.getElementById("button");

    // 画像選択時に表示するラベルを取得
    const chatImageLabel = document.getElementById("chat-image-label");

    // 画像が選択されたときに実行
    imageInput.addEventListener("change", function () {

      // 画像ファイルが1つ以上選択されているか確認
      if (imageInput.files.length > 0) {

        // ラベルの表示を変更
        chatImageLabel.textContent = "画像が添付されています";
      }
    });

    // 送信ボタンがクリックされたときに実行
    button.addEventListener("click", function () {

      // 入力されたメッセージ本文を取得
      const content = input.value;

      // 選択されている画像ファイルを取得
      // 画像がなければundefined
      const file = imageInput.files[0];

      // 画像が選択されている場合
      if (file) {

        // ファイルを読み込むためのFileReaderを作成
        const reader = new FileReader();

        // ファイルの読み込みが完了したときに実行
        reader.onload = function (event) {

          // 読み込んだデータからBase64部分だけを取得
          //
          // data:image/jpeg;base64,XXXXXX
          //                    ↑
          //                    この部分を取得
          const base64String = event.target.result.split(",")[1];

          // メッセージ本文と画像をRails側へ送信
          App.room.speak(content, base64String);
        };

        // 画像をBase64形式として読み込む
        reader.readAsDataURL(file);

      } else {

        // 画像がない場合はメッセージ本文だけ送信
        App.room.speak(content);
      }

      // 送信後にメッセージ入力欄を空にする
      input.value = "";

      // 選択した画像をリセットする
      imageInput.value = "";

      // 画像選択ラベルを元に戻す
      chatImageLabel.textContent = "画像";
    });
  });
}
